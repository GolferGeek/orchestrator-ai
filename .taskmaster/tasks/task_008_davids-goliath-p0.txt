```vue
<!-- apps/client/src/components/RunMetadataPanel.vue -->
<template>
  <div class="run-metadata-panel" v-if="hasMetadata">
    <div class="panel-header">
      <h3>Run Metadata</h3>
      <button @click="toggleExpanded" class="toggle-button">
        {{ expanded ? 'Collapse' : 'Expand' }}
      </button>
    </div>
    
    <div class="panel-content" v-if="expanded">
      <div class="metadata-item">
        <span class="label">Run ID:</span>
        <span class="value">{{ metadata.runId }}</span>
      </div>
      
      <div class="metadata-item">
        <span class="label">Provider:</span>
        <span class="value">{{ metadata.provider }}</span>
      </div>
      
      <div class="metadata-item">
        <span class="label">Model:</span>
        <span class="value">{{ metadata.model }}</span>
      </div>
      
      <div class="metadata-item">
        <span class="label">Tier:</span>
        <span class="value">{{ metadata.tier }}</span>
      </div>
      
      <div class="metadata-item">
        <span class="label">Duration:</span>
        <span class="value">{{ formatDuration(metadata.duration) }}</span>
      </div>
      
      <div class="metadata-item">
        <span class="label">Cost:</span>
        <span class="value">${{ formatCost(metadata.cost) }}</span>
      </div>
      
      <div class="metadata-item">
        <span class="label">Timestamp:</span>
        <span class="value">{{ formatTimestamp(metadata.timestamp) }}</span>
      </div>
    </div>
  </div>
</template>

<script>
export default {
  name: 'RunMetadataPanel',
  props: {
    metadata: {
      type: Object,
      default: () => ({})
    }
  },
  data() {
    return {
      expanded: false
    };
  },
  computed: {
    hasMetadata() {
      return this.metadata && this.metadata.runId;
    }
  },
  methods: {
    toggleExpanded() {
      this.expanded = !this.expanded;
    },
    formatDuration(duration) {
      return `${duration}ms`;
    },
    formatCost(cost) {
      return cost.toFixed(6);
    },
    formatTimestamp(timestamp) {
      return new Date(timestamp).toLocaleString();
    }
  }
};
</script>
```

2. Build `LocalModelStatus` dashboard:

```vue
<!-- apps/client/src/components/LocalModelStatus.vue -->
<template>
  <div class="local-model-status">
    <h2>Local Model Status</h2>
    
    <div class="tier-container" v-for="tier in tiers" :key="tier.name">
      <h3>{{ formatTierName(tier.name) }}</h3>
      
      <div class="models-grid">
        <div 
          v-for="model in tier.models" 
          :key="model.name"
          class="model-card"
          :class="getStatusClass(model.status)"
        >
          <div class="model-name">{{ model.name }}</div>
          <div class="model-status">{{ model.status }}</div>
          <div class="model-metrics" v-if="model.status === 'loaded'">
            <div>Response: {{ formatResponseTime(model.responseTime) }}</div>
            <div>Memory: {{ formatMemory(model.memoryUsage) }}</div>
          </div>
        </div>
      </div>
    </div>
    
    <div class="connection-status">
      <div class="status-indicator" :class="ollamaConnected ? 'connected' : 'disconnected'"></div>
      <span>Ollama Connection: {{ ollamaConnected ? 'Connected' : 'Disconnected' }}</span>
    </div>
    
    <div class="last-check">
      Last Updated: {{ lastChecked }}
    </div>
  </div>
</template>

<script>
export default {
  name: 'LocalModelStatus',
  data() {
    return {
      tiers: [],
      ollamaConnected: false,
      lastChecked: ''
    };
  },
  created() {
    this.fetchModelStatus();
    // Set up polling every 10 seconds
    this.pollingInterval = setInterval(this.fetchModelStatus, 10000);
  },
  beforeDestroy() {
    clearInterval(this.pollingInterval);
  },
  methods: {
    async fetchModelStatus() {
      try {
        const response = await fetch('/api/llm/local-models/status');
        const data = await response.json();
        
        this.tiers = data.tiers;
        this.ollamaConnected = data.ollamaConnected;
        this.lastChecked = new Date().toLocaleString();
      } catch (error) {
        console.error('Failed to fetch model status:', error);
        this.ollamaConnected = false;
      }
    },
    formatTierName(name) {
      return name.split('-').map(word => 
        word.charAt(0).toUpperCase() + word.slice(1)
      ).join(' ');
    },
    getStatusClass(status) {
      return {
        'status-loaded': status === 'loaded',
        'status-loading': status === 'loading',
        'status-error': status === 'error',
        'status-unavailable': status === 'unavailable'
      };
    },
    formatResponseTime(time) {
      return `${time}ms`;
    },
    formatMemory(bytes) {
      const gb = bytes / (1024 * 1024 * 1024);
      return `${gb.toFixed(1)} GB`;
    }
  }
};
</script>
```
