describe('End-to-end LLM Service Flow', () => {

    const agent = new TestAgent(llmService);
    
    // Mock Ollama to ensure local routing

    expect(result.runMetadata.tier).toBe('local');
    expect(result.runMetadata.provider).toBe('ollama');
    expect(result.runMetadata.runId).toBeDefined();
    expect(result.response).toBeDefined();
  });
  
  test('Falls back to external provider when local unavailable', async () => {
    // Setup test agent with LLMService
    const agent = new TestAgent(llmService);
    
    // Mock Ollama to force fallback
    mockOllamaAvailable(false);
    
    // Execute agent request
    const result = await agent.executeRequest('test prompt');
    
    // Verify fallback
    expect(result.runMetadata.tier).toBe('external');
    expect(['openai', 'anthropic']).toContain(result.runMetadata.provider);
    expect(result.runMetadata.runId).toBeDefined();
    expect(result.response).toBeDefined();
  });
});
