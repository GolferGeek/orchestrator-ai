```typescript
// Before migration
import { OpenAI } from 'openai';

class MetricsAgent {
  private openai: OpenAI;
  
  constructor() {
    this.openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY
    });
  }
  
  async analyzeMetrics(data: any) {
    const response = await this.openai.chat.completions.create({
      model: 'gpt-4',
      messages: [
        { role: 'system', content: 'You are a financial metrics analyst.' },
        { role: 'user', content: `Analyze these metrics: ${JSON.stringify(data)}` }
      ]
    });
    
    return response.choices[0].message.content;
  }
}

// After migration
import { LLMService } from '../../../llms/llm.service';

class MetricsAgent {
  constructor(private llmService: LLMService) {}
  
  async analyzeMetrics(data: any) {
    const result = await this.llmService.generateResponse(
      `Analyze these metrics: ${JSON.stringify(data)}`,
      {
        systemPrompt: 'You are a financial metrics analyst.',
        dataClass: 'financial', // Set appropriate data classification
        policyProfile: 'standard'
      }
    );
    
    return result.response;
  }
}
```
