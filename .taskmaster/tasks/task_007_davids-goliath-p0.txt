// apps/api/src/llms/secret-redaction.service.ts
import { Injectable, LoggerService } from '@nestjs/common';

@Injectable()
export class SecretRedactionService {
  private patterns: RegExp[] = [
    // API key patterns
    /sk-[a-zA-Z0-9]{32,}/g, // OpenAI API keys
    /sk-ant-[a-zA-Z0-9]{32,}/g, // Anthropic API keys
    
    // Auth token patterns
    /Bearer\s+[a-zA-Z0-9._-]+/gi,
    /Authorization:\s*['"]?Bearer\s+[a-zA-Z0-9._-]+['"]?/gi,
    
    // Password patterns
    /password['"]?\s*[:=]\s*['"]([^'"]+)['"]?/gi,
    
    // Environment variable patterns
    /([A-Z_]+_KEY|[A-Z_]+_SECRET|[A-Z_]+_PASSWORD)\s*[:=]\s*['"]?([^'"\s]+)['"]?/gi
  ];
  
  private isProduction: boolean;
  
  constructor() {
    this.isProduction = process.env.NODE_ENV === 'production';
  }
  
  redact(text: string): string {
    if (!text) return text;
    
    let redacted = text;
    for (const pattern of this.patterns) {
      redacted = redacted.replace(pattern, '[REDACTED]');
    }
    
    return redacted;
  }
  
  redactObject(obj: any): any {
    if (!obj) return obj;
    if (typeof obj !== 'object') return obj;
    
    const result = Array.isArray(obj) ? [] : {};
    
    for (const [key, value] of Object.entries(obj)) {
      // Skip redaction for safe keys
      if (['id', 'runId', 'timestamp', 'level'].includes(key)) {
        result[key] = value;
        continue;
      }
      
      // Redact sensitive keys entirely
      if (['password', 'secret', 'token', 'key', 'authorization'].includes(key.toLowerCase())) {
        result[key] = '[REDACTED]';
        continue;
      }
      
      // Recursively redact objects
      if (typeof value === 'object' && value !== null) {
        result[key] = this.redactObject(value);
        continue;
      }
      
      // Redact strings
      if (typeof value === 'string') {
        result[key] = this.redact(value);
        continue;
      }
      
      // Pass through other values
      result[key] = value;
    }
    
    return result;
  }
  
  shouldLogBody(): boolean {
    return !this.isProduction;
  }
  
  log(level: 'info' | 'warn' | 'error', message: string, context: any): void {
    const logEntry = {
      runId: context.runId || 'unknown',
      timestamp: new Date().toISOString(),
      level,
      message: this.redact(message)
    };
    
    // In production, don't log request/response bodies
    if (this.isProduction && context.body) {
      delete context.body;
    }
    
    // Redact any remaining context data
    const safeContext = this.redactObject(context);
    
    // Log the entry (implementation depends on logging library)
    console.log(JSON.stringify({ ...logEntry, context: safeContext }));
  }
}
