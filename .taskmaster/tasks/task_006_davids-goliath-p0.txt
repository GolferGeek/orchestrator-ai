// apps/api/src/llms/centralized-routing.service.ts
import { Injectable } from '@nestjs/common';
import { LocalModelStatusService } from './local-model-status.service';

export interface RoutingDecision {
  provider: string;
  model: string;
  isLocal: boolean;
  modelTier?: string;
  fallbackUsed: boolean;
}

export interface LLMRequest {
  prompt: string;
  options?: any;
}

@Injectable()
export class CentralizedRoutingService {
  constructor(
    private localModelService: LocalModelStatusService,
    private databaseService: any // Inject your database service
  ) {}
  
  async determineRoute(prompt: string, options: any = {}): Promise<RoutingDecision> {
    // If specific provider/model is requested, honor it
    if (options.provider && options.model) {
      return {
        provider: options.provider,
        model: options.model,
        isLocal: options.provider === 'ollama',
        fallbackUsed: false
      };
    }
    
    // 1. Analyze request complexity
    const complexity = this.analyzeComplexity({ prompt, options });
    
    // 2. Select appropriate tier based on complexity
    const tier = this.selectTierForComplexity(complexity);
    
    // 3. Check if local models in the tier are available
    const availableModels = await this.getAvailableLocalModels(tier);
    
    if (availableModels.length > 0) {
      // 4a. Select best local model from tier
      const selectedModel = this.selectBestModel(availableModels, { prompt, options });
      return {
        provider: 'ollama',
        model: selectedModel.name,
        isLocal: true,
        modelTier: tier,
        fallbackUsed: false
      };
    } else {
      // 4b. Fall back to external provider
      return this.getExternalFallback(tier, { prompt, options });
    }
  }
  
  private analyzeComplexity(request: LLMRequest): 'simple' | 'medium' | 'complex' {
    const { prompt } = request;
    
    // Simple heuristics for complexity analysis
    const wordCount = prompt.split(/\s+/).length;
    const sentenceCount = prompt.split(/[.!?]+/).length;
    
    if (wordCount < 20) {
      return 'simple';
    } else if (wordCount < 100) {
      return 'medium';
    } else {
      return 'complex';
    }
    
    // More sophisticated analysis could include:
    // - Keyword detection for complex tasks
    // - Sentiment analysis
    // - Topic classification
  }
  
  private selectTierForComplexity(complexity: 'simple' | 'medium' | 'complex'): string {
    switch (complexity) {
      case 'simple':
        return 'ultra-fast';
      case 'medium':
        return 'general';
      case 'complex':
        return 'fast-thinking';
    }
  }
  
  private async getAvailableLocalModels(tier: string): Promise<any[]> {
    // 1. Query database for models in the specified tier
    const models = await this.databaseService.query(`
      SELECT * FROM llm_models 
      WHERE is_local = TRUE AND model_tier = $1
      ORDER BY loading_priority DESC
    `, [tier]);
    
    // 2. Check with localModelService if they're currently loaded
    const availableModels = [];
    
    for (const model of models) {
      const health = await this.localModelService.checkModelHealth(model.name);
      if (health.available) {
        availableModels.push({
          ...model,
          health
        });
      }
    }
    
    return availableModels;
  }
  
  private selectBestModel(models: any[], request: LLMRequest): any {
    // Select best model based on priority and request characteristics
    // For now, simply return the highest priority model
    return models[0];
  }
  
  private getExternalFallback(tier: string, request: LLMRequest): RoutingDecision {
    // Map local tier to appropriate external provider/model
    switch (tier) {
      case 'fast-thinking':
        return {
          provider: 'openai',
          model: 'gpt-4',
          isLocal: false,
          modelTier: 'external-advanced',
          fallbackUsed: true
        };
      case 'general':
        return {
          provider: 'openai',
          model: 'gpt-3.5-turbo',
          isLocal: false,
          modelTier: 'external-standard',
          fallbackUsed: true
        };
      case 'ultra-fast':
        return {
          provider: 'openai',
          model: 'gpt-3.5-turbo-instruct',
          isLocal: false,
          modelTier: 'external-fast',
          fallbackUsed: true
        };
      default:
        return {
          provider: 'openai',
          model: 'gpt-3.5-turbo',
          isLocal: false,
          fallbackUsed: true
        };
    }
  }
}

// apps/api/src/llms/model-preloader.service.ts
import { Injectable, OnModuleInit } from '@nestjs/common';
import { LocalModelStatusService } from './local-model-status.service';

@Injectable()
export class ModelPreloaderService implements OnModuleInit {
  constructor(
    private databaseService: any, // Inject your database service
    private localModelService: LocalModelStatusService
  ) {}
  
  async onModuleInit() {
    // Preload models when the application starts
    await this.preloadHighPriorityModels();
  }
  
  async preloadHighPriorityModels(): Promise<void> {
    const models = await this.databaseService.query(`
      SELECT * FROM llm_models 
      WHERE is_local = TRUE
      ORDER BY loading_priority DESC
      LIMIT 5
    `);
    
    for (const model of models) {
      try {
        console.log(`Preloading model: ${model.name}`);
        await this.localModelService.ensureModelLoaded(model.name);
        await this.databaseService.query(`
          UPDATE llm_models
          SET is_currently_loaded = TRUE
          WHERE name = $1
        `, [model.name]);
      } catch (error) {
        console.error(`Failed to preload model ${model.name}:`, error);
      }
    }
  }
  
  async ensureThreeTierModelsLoaded(): Promise<void> {
    // Ensure at least one model from each tier is loaded
    const tiers = ['fast-thinking', 'ultra-fast', 'general'];
    
    for (const tier of tiers) {
      const models = await this.databaseService.query(`
        SELECT * FROM llm_models 
        WHERE is_local = TRUE AND model_tier = $1
        ORDER BY loading_priority DESC
        LIMIT 1
      `, [tier]);
      
      if (models.length > 0) {
        await this.localModelService.ensureModelLoaded(models[0].name);
      }
    }
  }
}
