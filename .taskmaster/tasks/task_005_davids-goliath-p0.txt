// apps/api/src/llms/provider-config.service.ts
import { Injectable } from '@nestjs/common';

export interface ProviderConfig {
  timeout: number;
  retries: number;
  headers: Record<string, string>;
  noTrainFlags: Record<string, any>;
}

@Injectable()
export class ProviderConfigService {
  getConfig(provider: string): ProviderConfig {
    switch (provider.toLowerCase()) {
      case 'openai':
        return {
          timeout: 30000,
          retries: 2,
          headers: {
            'X-No-Store': 'true'
          },
          noTrainFlags: {
            use_context: false
          }
        };
      case 'anthropic':
        return {
          timeout: 30000,
          retries: 2,
          headers: {
            'X-API-Policy-Compliance': 'no-train'
          },
          noTrainFlags: {
            policyCompliance: 'no-train'
          }
        };
      case 'ollama':
        return {
          timeout: 60000, // Local models may need longer timeout
          retries: 1,
          headers: {},
          noTrainFlags: {}
        };
      default:
        return {
          timeout: 30000,
          retries: 2,
          headers: {},
          noTrainFlags: {}
        };
    }
  }
  
  createAdapter(provider: string): ProviderAdapter {
    const config = this.getConfig(provider);
    
    switch (provider.toLowerCase()) {
      case 'openai':
        return new OpenAIAdapter(config);
      case 'anthropic':
        return new AnthropicAdapter(config);
      case 'ollama':
        return new OllamaAdapter(config);
      default:
        throw new Error(`Unsupported provider: ${provider}`);
    }
  }
  
  getProviderOptions(provider: string, baseOptions: any = {}): any {
    const config = this.getConfig(provider);
    const noTrainFlags = config.noTrainFlags;
    
    return {
      ...baseOptions,
      timeout: config.timeout,
      ...noTrainFlags
    };
  }
}
